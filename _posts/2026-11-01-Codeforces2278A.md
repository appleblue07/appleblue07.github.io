# 코드포스 2278A 문제에 대한 저의 풀이입니다

[Codeforces 2278A 문제원문](https://codeforces.com/problemset/problem/2278/A)


이 문제는 처음 보고 웃음이 나왔는데 영단어 몰라서 못 풀었습니다. ㅋㅋㅋㅋ
<br><br>

분모와 분자에 같은 알파벳이 있으면 약분 할 수 있는데 남는게 six면 YES 아니면 NO를 출력하면 됩니다. 조건만 조심하면 단순 구현문제입니다.


<br>
<br>
<br>
<br>
<br>
아래는 정답코드입니다.

```cpp

#include <bits/stdc++.h>
using namespace std;

void func(){
        int n, m;
        scanf("%d%d", &n, &m);
        char a[n+1], b[m+1];

        scanf(" %s %s", a, b);
        int idx = 0;
        vector<char> t(1, '\0');

        for(int i = 0 ; i < n ; i++){
                t.push_back(a[i]);
        }

        for(int i = 0 ; i < m ; i++){
                for(int j = 0 ; j < n ; j++){
                        if(b[i] == t[j+1]){
                                t[j+1] = '\0';
                                b[i] = '\0';
                                break;
                        }
                }

                if(b[i] != '\0'){
                        printf("NO\n");
                        return;
                }
        }

        int c1 = 0, c2 = 0, c3 = 0;
        for(int i = 0 ; i <= n ; i++){
                if(t[i] == '\0') continue;

                if(t[i] == 's') c1++;
                if(t[i] == 'i') c2++;
                if(t[i] == 'x') c3++;

                if(t[i] != '\0' &&
                        t[i] != 's' &&
                        t[i] != 'i' &&
                        t[i] != 'x'){
                        printf("NO\n");
                        return;
                }
        }

        //printf("%d %d %d\n", c1, c2, c3);
        if(c1 == 1 && c2 == 1 && c3 == 1) printf("YES\n");
        else printf("NO\n");

        return;
}

int main(void){
        int t;
        scanf("%d", &t);
        for(int i = 0 ; i < t ; i++) func();
        return 0;
}

```
