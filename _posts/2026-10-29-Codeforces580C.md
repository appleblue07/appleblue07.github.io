# 코드포스 580C 문제에 대한 저의 풀이입니다

[Codeforces 580C 문제원문](https://codeforces.com/problemset/problem/580/C) 

트리 그래프가 주어졌을 때 루트에서 리프 노드까지 가는 경로 중 고양이가 연속적(consecutive)으로  m개 이하인 경우를 모두 찾으면 되는 문제입니다.




dfs 와 bfs 풀이 모두 가능합니다.
(저는 dfs에서 한 번 방문한 노드를 다시 확인할 필요가 없어 백트레킹을 해도 n의 시간이 걸린다는 걸 생각하지 못해 bfs로 풀었습니다)

아래는 저의 정답 코드입니다


```cpp

#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> v(100001);

int main(){
	int n, m;
	scanf("%d%d", &n, &m);
	vector<int> a(n+1);

	for(int i = 1 ; i <= n ; i++){
		scanf("%d", &a[i]);
	}
	
	int s, e;
	for(int i = 0 ; i < n-1; i++){
		scanf("%d%d", &s, &e);
		v[s].push_back(e);
		v[e].push_back(s);
	}
	
	//bfs

	queue<pair<int, int>> q;
	q.push({1, a[1]});
	
	pair<int, int> t;
	vector<int> visited(n+1, 0);

	int ans = 0;
	
	while(!q.empty()){
		t = q.front();
		q.pop();
		
		if(t.second > m || visited[t.first] == 1) continue;
		
		visited[t.first] = 1;

		if(t.first != 1 && v[t.first].size() == 1) {
			//printf("node : %d, cat : %d, t : %d\n", t.first, t.second, a[t.second]);
			ans++;
			continue;
		}

		
		
		for(auto u : v[t.first]){
			//다음 노드를 추가하기
			//다음 노드에 고양이가 있으면 전에꺼에서 더해서 연속된 수를 두번쨰에 저장하기
			//없으면 연속이 끊어지므로 0 저장
			if(a[u] == 1) q.push({u, t.second + a[u]});
			else q.push({u, 0});
		}
	}
	
	printf("%d\n", ans);
	return 0;
}

```



여담으로 구현하다보니까 오류 추적하기 헷갈리더라고요 관계그래프를 나타낼떄 두 점을 서로 이어야 한다던가 시작점은 0이아니라 1이라는 걸 못봐서 답답한 상황이 있었습니다. 인덱스나 변수 추적하는 기본기 연습을 더 해야겠네요.
