### AtCoder ABC404 c번 문제 해설입니다.

문제 자체는 단순합니다. 그냥 정의를 보고 사이클 그래프인지 판단하면 됩니다. 저는 두 가지 방법으로 풀었는데(~~시간이 다 지나고~~) 둘다 AC 받았습니다.

첫번째 방식은 dfs를 활용해서 사이클 그래프인지 판단을 기준으로 점을 dfs 하면 됩니다<br>
아래는 정답코드입니다<br>
<br>

```cpp
#include <bits/stdc++.h>
using namespace std;

int main(void){

        int n, m;

        scanf("%d %d", &n, &m);
        vector<vector<int>> g(n+1);


        int s, e;
        for(int i = 0 ; i < m ; i++){
                scanf("%d%d", &s, &e);
                g[s].push_back(e);
                g[e].push_back(s);
        }

        vector<int> vis(n+1, 0);
        // 이전 노드까지 저장해두고 사이클이 이상하게 되는지 판단하려 했는데 다시 생각해보니 필요 없네요.....
        stack<pair<int, int>> p; 
        
        p.push({s, e});

        pair<int, int> t;

        while(!p.empty()){

                t = p.top();
                p.pop();

                if(vis[t.first]) {
                        // 모든 노드를 방문했는지 확인
                        for(int i = 1 ; i <= n ; i++) if(vis[i] == 0){
                          printf("No\n");
                          return 0;
                        }
                        printf("Yes\n");
                        
                        return 0;
                }

                vis[t.first] = 1;

                int r = 0;

                for(auto u : g[t.first]){
                        r++;
                        if(r > 2) { // 차수가 2개가 넘으면 사이클 그래프가 될 수 없음
                                printf("No\n");
                                return 0;
                        }
                        if(t.second != u) p.push({u, t.first});
                }
        }

        printf("No\n");
        return 0;
}
```

두번째 방법은 모든 정점의 차수가 2이고 모두 연결되어있는지 판단하는 방법입니다.
<br>
아래는 그렇게 구현한 정답 코드입니다.
<br><br>

``` cpp
#include <bits/stdc++.h>
using namespace std;

int main(void){

        int n, m;

        scanf("%d %d", &n, &m);
        vector<vector<int>> g(n+1);


        int s, e;
        for(int i = 0 ; i < m ; i++){
                scanf("%d%d", &s, &e);
                g[s].push_back(e);
                g[e].push_back(s);
        }

        for(int i = 1 ; i <= n ; i++){
                if(g[i].size() != 2){ // 차수가 2가 아니면 사이클 그래프가 아님
                        printf("No\n");
                        return 0;
                }
        }

        s = 1;

        vector<int> v(n+1, 0);

        for(int i = 0 ; i < n ; i++){
                if(v[s] && i != n-1) {
                    //이미 방문한 노드라면 사이클이 생긴것이므로 길이가 n-1인지 확인하면 됨
                    // 여기서 i는 방문한 노드의 수가 됨
                        printf("No\n");
                        return 0;
                }
                v[s] = 1;
                
                //위 조건 식에 의해 차수가 2개인 점들만 있음
                //그 중 방문하지 않은 점( 바로 직전 노드 를 제외) 확인
                if(v[g[s][0]]) e = g[s][1]; 
                else e = g[s][0];

                s = e;
        }

        //위 조건을 모두 뚫으면 사이클 그래프
        printf("Yes\n");
        return 0;
}
```
